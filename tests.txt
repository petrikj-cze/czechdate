console.log("Dnes je " + czechDate()) //defaultní výpis [Dnes je 11. srpna 2026]

console.log("Dnes je " + czechDate("nominativ")) //defaultní výpis [Dnes je 11. srpna 2026]

console.log("Dnes je měsíc " + czechDate("nominativ", "M")) //pouze měsíc v nominativu [Dnes je měsíc srpen]

console.log("Příští kontrola bude v " + czechDate("locativ", "M")) //pouze měsíc v locativu [Příští kontrola bude v srpnu]

//
// S UŽIVATELSKY NASTAVENÝM DATUMEM VE FORMÁTU DD.MM.YYYY nebo D.M.YYYY nebo DD.M.YYYY nebo D.MM.YYYY
